## scripts/analyze_r1.py
```python
import csv, json, os

src = r"C:\Users\13829\Documents\deepseek-harness\default-workspace\veriflow\artifacts\T-08\data\raw.csv"
dst = r"C:\Users\13829\Documents\deepseek-harness\default-workspace\veriflow\artifacts\T-08\result.json"

def to_mg(v, unit):
    u = (unit or "").strip().lower()
    if u in ("mg", ""):
        return v
    if u == "ug":
        return v / 1000.0
    if u == "g":
        return v * 1000.0
    if u == "ng":
        return v / 1000000.0
    return v

rows = []
with open(src, "r", encoding="utf-8-sig", newline="") as f:
    reader = csv.DictReader(f)
    for r in reader:
        rows.append(r)

vals_a, vals_b = [], []
for r in rows:
    raw = r.get("value")
    if raw is None:
        continue
    s = str(raw).strip()
    if s == "":
        continue
    try:
        v = float(s)
    except ValueError:
        continue
    g = (r.get("group") or "").strip()
    v = to_mg(v, r.get("unit"))
    if g == "A":
        vals_a.append(v)
    elif g == "B":
        vals_b.append(v)

n_a = len(vals_a)
n_b = len(vals_b)
mean_a = sum(vals_a) / n_a if n_a else None
mean_b = sum(vals_b) / n_b if n_b else None
diff = (mean_b - mean_a) if (n_a and n_b) else None

out = {"mean_a": mean_a, "mean_b": mean_b, "diff": diff, "n_a": n_a, "n_b": n_b}
os.makedirs(os.path.dirname(dst), exist_ok=True)
with open(dst, "w", encoding="utf-8") as f:
    json.dump(out, f, ensure_ascii=False, indent=2)
print(json.dumps(out, ensure_ascii=False))
```
