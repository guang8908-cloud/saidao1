## scripts/analyze_r1.py
```python
import csv, json, os

src = r"C:\Users\13829\Documents\deepseek-harness\default-workspace\veriflow\artifacts\T-09\data\raw.csv"
dst = r"C:\Users\13829\Documents\deepseek-harness\default-workspace\veriflow\artifacts\T-09\result.json"

UNIT_TO_BASE = {
    "ug": 1.0,
    "mcg": 1.0,
    "µg": 1.0,
    "μg": 1.0,
    "mg": 1000.0,
    "g": 1000000.0,
    "ng": 0.001,
}

def parse_value(raw):
    if raw is None:
        return None
    s = raw.strip()
    if s == "":
        return None
    try:
        return float(s)
    except ValueError:
        return None

def normalize_unit(u):
    if u is None:
        return None
    s = u.strip().lower()
    if s == "":
        return None
    return s

rows = []
with open(src, "r", encoding="utf-8-sig", newline="") as f:
    reader = csv.DictReader(f)
    for r in reader:
        rows.append(r)

# 先确定统一目标单位：取出现过的单位中，按基准换算后选择出现最多的量级
unit_counts = {}
for r in rows:
    v = parse_value(r.get("value"))
    if v is None:
        continue
    u = normalize_unit(r.get("unit"))
    if u is None or u not in UNIT_TO_BASE:
        continue
    unit_counts[u] = unit_counts.get(u, 0) + 1

target_unit = None
if unit_counts:
    target_unit = max(unit_counts.items(), key=lambda kv: kv[1])[0]

groups = {"A": [], "B": []}
for r in rows:
    g = (r.get("group") or "").strip()
    if g not in groups:
        continue
    v = parse_value(r.get("value"))
    if v is None:
        continue
    u = normalize_unit(r.get("unit"))
    if u is None or u not in UNIT_TO_BASE:
        continue
    if target_unit is not None:
        v = v * UNIT_TO_BASE[u] / UNIT_TO_BASE[target_unit]
    groups[g].append(v)

n_a = len(groups["A"])
n_b = len(groups["B"])
mean_a = sum(groups["A"]) / n_a if n_a > 0 else None
mean_b = sum(groups["B"]) / n_b if n_b > 0 else None
diff = (mean_b - mean_a) if (mean_a is not None and mean_b is not None) else None

result = {
    "mean_a": mean_a,
    "mean_b": mean_b,
    "diff": diff,
    "n_a": n_a,
    "n_b": n_b,
    "unit": target_unit,
}

os.makedirs(os.path.dirname(dst), exist_ok=True)
with open(dst, "w", encoding="utf-8") as f:
    json.dump(result, f, ensure_ascii=False, indent=2)

print(json.dumps(result, ensure_ascii=False))
```
