## scripts/analyze_r1.py
```python
import csv, json, os

src = r"C:\Users\13829\Documents\deepseek-harness\default-workspace\veriflow\artifacts\T-04\data\clean.csv"
dst = r"C:\Users\13829\Documents\deepseek-harness\default-workspace\veriflow\artifacts\T-04\result.json"

def to_mg(v, unit):
    u = (unit or "").strip().lower()
    if u in ("mg", ""):
        return v
    if u == "ug":
        return v / 1000.0
    if u == "g":
        return v * 1000.0
    if u == "ng":
        return v / 1000000.0
    return v

sums = {"A": 0.0, "B": 0.0}
counts = {"A": 0, "B": 0}

with open(src, "r", encoding="utf-8-sig", newline="") as f:
    reader = csv.DictReader(f)
    for row in reader:
        g = (row.get("group") or "").strip()
        if g not in ("A", "B"):
            continue
        raw = row.get("value")
        if raw is None:
            continue
        s = raw.strip()
        if s == "":
            continue
        try:
            v = float(s)
        except ValueError:
            continue
        v = to_mg(v, row.get("unit"))
        sums[g] += v
        counts[g] += 1

n_a, n_b = counts["A"], counts["B"]
mean_a = sums["A"] / n_a if n_a else None
mean_b = sums["B"] / n_b if n_b else None
diff = (mean_b - mean_a) if (n_a and n_b) else None

result = {"mean_a": mean_a, "mean_b": mean_b, "diff": diff, "n_a": n_a, "n_b": n_b}

os.makedirs(os.path.dirname(dst), exist_ok=True)
with open(dst, "w", encoding="utf-8") as f:
    json.dump(result, f, ensure_ascii=False, indent=2)

print(json.dumps(result, ensure_ascii=False))
```
