def to_mg(v, unit):
    u = (unit or "").strip().lower()
    if u in ("mg", ""):
        return v
    if u == "ug":
        return v / 1000.0
    if u == "g":
        return v * 1000.0
    if u == "ng":
        return v / 1000000.0
    return v

groups = {"A": [], "B": []}
with open(src, "r", encoding="utf-8-sig", newline="") as f:
    reader = csv.DictReader(f)
    for row in reader:
        raw = row.get("value")
        if raw is None:
            continue
        s = raw.strip()
        if s == "":
            continue
        try:
            val = float(s)
        except ValueError:
            continue
        g = (row.get("group") or "").strip()
        if g not in groups:
            continue
        groups[g].append(to_mg(val, row.get("unit")))

n_a = len(groups["A"])
n_b = len(groups["B"])
mean_a = sum(groups["A"]) / n_a if n_a else None
